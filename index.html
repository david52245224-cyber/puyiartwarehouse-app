Google 日曆 Apps Script 連線設定

💡 說明：本程式支援直接「讀取日曆」、「打 V / 取消 V」以及「修改預約日期與時間」。請確認您的 Google Apps Script 包含最新的 doPost 代碼並已發布新版本。
Code.gs 完整代碼 (支援讀取 + 打 V + 改期) 
function doGet() {
  var now = new Date();
  now.setHours(0, 0, 0, 0); 
  var end = new Date(now.getTime() + (14 * 24 * 60 * 60 * 1000));
  
  var calendars = CalendarApp.getAllCalendars();
  var allEvents = [];
  
  calendars.forEach(function(cal) {
    try {
      var events = cal.getEvents(now, end);
      events.forEach(function(e) {
        var title = e.getTitle();
        if (title.indexOf("樸藝小倉庫") !== -1 && title.indexOf("預約") !== -1) {
          allEvents.push({
            id: e.getId(),
            calendarId: cal.getId(),
            calendarName: cal.getName(),
            title: title,
            start: e.getStartTime().toISOString(),
            end: e.getEndTime().toISOString(),
            description: e.getDescription() || "",
            location: e.getLocation() || ""
          });
        }
      });
    } catch(err) {}
  });
  
  allEvents.sort(function(a, b) {
    return new Date(a.start) - new Date(b.start);
  });
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: allEvents }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    var calendarId = payload.calendarId || "primary";
    var eventId = payload.eventId;
    
    var cal = CalendarApp.getCalendarById(calendarId) || CalendarApp.getDefaultCalendar();
    var event = cal.getEventById(eventId);
    if (!event) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "找不到該日曆行程" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "toggleConfirm") {
      var currentTitle = event.getTitle();
      var newTitle = currentTitle;
      if (payload.confirmed) {
        if (!/[\s\(]?[vVＶ✓✔]/i.test(currentTitle)) {
          newTitle = currentTitle.trim() + " V";
        }
      } else {
        newTitle = currentTitle.replace(/\s*[vVＶ✓✔]\s*$/i, "").trim();
      }
      event.setTitle(newTitle);
      return ContentService.createTextOutput(JSON.stringify({ status: "success", newTitle: newTitle }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "updateEventTime") {
      var newStart = new Date(payload.newStartTime);
      var newEnd = new Date(payload.newEndTime);
      event.setTime(newStart, newEnd);
      if (payload.note && payload.note.trim() !== "") {
        var prevDesc = event.getDescription() || "";
        event.setDescription(prevDesc + "\n[改期備註 " + Utilities.formatDate(new Date(), "GMT+8", "MM/dd HH:mm") + "] " + payload.note);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "時間已成功更新" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "未知操作" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
